Topological Sorting
Aug 30, 2024

Overview

In computer science, a topological sort or topological ordering of a directed graph is a linear ordering of its vertices. For instance, the vertices of the graph may represent tasks to be performed, and the edges may represent constraints that one task ust be performed before another; in this application, a topological ordering is just a valid sequence for the tasks.

A topological ordering is possible if and only if the grapg has no directed cycles,that is, if it is a directed acyclic graph (DAG).

Algorithms

Kahn’s algorithm

Time complexity O(V+E), Space O(V)

  1. convert input data to a directed graph.
  2. use an array to record the incoming edges of vertices.
  3. add the nodes which has no incoming edge into a queue.
  4. use the queue to traverse the graph.
  5. repeat 3 and 4 tile the queue is empty.

DFS

Time complexity O(V+E), Space O(V+E)

  1. convert input data to a directed graph.
  2. use an array to mark vertices’s states, each vertex has three states including initial, temporary and permanent. If we visite a temporary vertex then return false(cycle detect ), if we visite a permanent vertex, then return true. (to avoid repeated traversal)
  3. add a permanent vertex into the top of a stack.
  4. repeat 3 and 4 tile all vertices was visited.

//Kahn's algorithm
class Solution {
public:
    bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
        vector<int>adj[numCourses];
        vector<int>indegree(numCourses,0);
        int ans=0;
        for(auto x:prerequisites){
            adj[x[1]].push_back(x[0]); //construct a graph (step 1)
            indegree[x[0]]++; //step2
        }
        queue<int>q;
        for(int i=0;i<numCourses;i++){
            if(indegree[i]==0) q.push(i); //step 3
        }
        while(!q.empty()){
            int now=q.front();
            ans++;
            q.pop();
            for(auto x:adj[now]){
                indegree[x]--;
                if(indegree[x]==0) q.push(x);
            }
        }
        return ans==numCourses;
    }
};

//DFS
class Solution {
public:
    bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
        vector<vector<int>>adj(numCourses);
        vector<int>mark(numCourses,0);
        bool rst;
        int ans=0;
        for(auto x:prerequisites){
            adj[x[1]].push_back(x[0]);
        }
        //mark[i]=0 initial
        //mark[i]=1 temporary
        //mark[i]=2 permanent
        for(int i=0;i<numCourses;i++){
            if(mark[i]<2){
                rst=dfs(i,mark,adj);
                if(!rst) return false;
            }
        }
        return true;
    }
    bool dfs(int i,vector<int>&mark,vector<vector<int>>&adj){
        if(mark[i]==2) return true;
        if(mark[i]==1) return false; //detect cycle
        mark[i]=1;
        for(auto x:adj[i]){
            if(!dfs(x,mark,adj)) return false;
        }
        mark[i]=2;
        return true;
    }
};

class Solution {
public:
    vector<int> findOrder(int numCourses, vector<vector<int>>& prerequisites) {
        vector<int>adj[numCourses];
        vector<int>indegree(numCourses,0);
        vector<int>ans;
        for(auto x:prerequisites){
            adj[x[1]].push_back(x[0]);
            indegree[x[0]]++;
        }
        queue<int>q;
        for(int i=0;i<numCourses;i++){
            if(indegree[i]==0) q.push(i);
        }
        while(!q.empty()){
            int now=q.front();
            ans.push_back(now);
            q.pop();
            for(auto x:adj[now]){
                indegree[x]--;
                if(indegree[x]==0) q.push(x);
            }
        }
        if(ans.size()!=numCourses){
            return {};
        }
        return ans;
    }
};

class Solution {
public:
    vector<vector<int>> buildMatrix(int k, vector<vector<int>>& rowConditions, vector<vector<int>>& colConditions) {
        vector<vector<int>>rowgraph(k+1),colgraph(k+1);
        for(auto& rc:rowConditions){
            rowgraph[rc[0]].push_back(rc[1]);
        }
        for(auto& cc:colConditions){
            colgraph[cc[0]].push_back(cc[1]);
        }
        vector<int> roworder=topoSort(rowgraph,k);
        vector<int> colorder=topoSort(colgraph,k);

        if(roworder.empty()||colorder.empty()) return{};
        vector<vector<int>>result(k,vector<int>(k,0));
        unordered_map<int,int>rowMap,colMap;
        for(int i=0;i<k;++i){
        // i represents position, order[i] represents number
            rowMap[roworder[i]]=i;
            colMap[colorder[i]]=i;
        }
        for(int i=1;i<=k;i++){
        //Map[i] represent i's position
            result[rowMap[i]][colMap[i]]=i;
        }
        return result;
    }
    
    //Kahn's algorithm
    vector<int> topoSort(vector<vector<int>> &graph, int k){
        vector<int>indegree(k+1,0),order;
        queue<int>q;
        for(int i=1;i<=k;i++){
            for(int j:graph[i]){
                ++indegree[j];
            }
        }
        for(int i=1;i<=k;i++){
            if(indegree[i]==0) q.push(i);
        }
        while(!q.empty()){
            int node=q.front();
            q.pop();
            order.push_back(node);
            for(int adj:graph[node]){
                if(--indegree[adj]==0) q.push(adj);
            }
        }
        return order.size()==k? order:vector<int>();
    }
};
